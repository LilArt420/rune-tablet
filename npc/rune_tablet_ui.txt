//===== Ragnarok Offline: rune-tablet =========================================
//= The Rune Tablet window (client/index.js) and its NPC in Prontera.
//=
//= The system itself is the server's Rune Tablet (npc/re/world_sync/
//= rune_tablet.txt, "Rune Tablet#ws"): its data arrays and the same character
//= variables (RT_R_<rune>, RT_L_<set>, RT_F_<set>, RT_W_<set>, RT_SET[],
//= RT_CNT[]) and set bonuses (F_RuneTabletBonus). This script only adds the
//= window: its NPC opens it, and @runetab answers it:
//=   @runetab <n> state                 -> r:<rune ids>;s:<set>=<level>/<fail bonus>,...
//=   @runetab <n> count <id>,<id>,...   -> <id>=<count>,...
//=   @runetab <n> register <rune id>    -> ok|<text> or err|<text>
//=   @runetab <n> upgrade <set id>      -> ok|success|<level>, ok|fail|<fail bonus>, err|<text>
//=   @runetab <n> decomplist            -> <item id>=<count>:<many ok>,...
//=   @runetab <n> decompose <id> <1|n>  -> ok|<item>:<qty>,... or err|<text>
//= Replies go out as "@@reply <n> <part>/<parts> <text>" lines.
//=============================================================================

prontera,193,189,4	script	Rune Tablet#rtui	4_RUNESTONE,{
	mes "[Rune Tablet]";
	mes "Here are your rune tablets: register runes in the book, build their sets and grow them stronger, or break items down into rune materials.";
	close2;
	dispbottom "@@event runetab open";
	end;

OnInit:
	bindatcmd "runetab", strnpcinfo(3) + "::OnReq", 0, 99;
	// the dialog version stays loaded (its login event re-applies the set bonuses), out of sight
	hideonnpc "Rune Tablet#ws";
	end;

S_Reply:
	// getarg(0): request number, getarg(1): text. Splits it into 200-character parts.
	.@text$ = getarg(1);
	.@len = getstrlen(.@text$);
	if (.@len == 0) {
		dispbottom "@@reply " + getarg(0) + " 1/1 ";
		return;
	}
	.@parts = (.@len + 199) / 200;
	for (.@p = 0; .@p < .@parts; .@p++)
		dispbottom "@@reply " + getarg(0) + " " + (.@p + 1) + "/" + .@parts + " " + substr(.@text$, .@p * 200, min(.@p * 200 + 199, .@len - 1));
	return;

S_Has:
	// getarg(0): "id,qty;id,qty". 1 when the player carries all of it.
	explode(.@row$, getarg(0), ";");
	for (.@i = 0; .@i < getarraysize(.@row$); .@i++) {
		explode(.@f$, .@row$[.@i], ",");
		if (countitem(atoi(.@f$[0])) < atoi(.@f$[1])) return 0;
	}
	return 1;

S_Take:
	explode(.@row$, getarg(0), ";");
	for (.@i = 0; .@i < getarraysize(.@row$); .@i++) {
		explode(.@f$, .@row$[.@i], ",");
		delitem atoi(.@f$[0]), atoi(.@f$[1]);
	}
	return;

S_Recount:
	// as the Rune Tablet's own: rebuild RT_SET/RT_CNT, hand out new rewards, refresh the bonus
	deletearray RT_SET;
	deletearray RT_CNT;
	.@n = 0;
	.@ns = getarraysize(getvariableofnpc(.s_id, "Rune Tablet#ws"));
	for (.@s = 0; .@s < .@ns; .@s++) {
		.@sid = getvariableofnpc(.s_id[.@s], "Rune Tablet#ws");
		deletearray .@slot$;
		explode(.@slot$, getvariableofnpc(.s_slot$[.@s], "Rune Tablet#ws"), ",");
		.@c = 0;
		for (.@i = 0; .@i < getarraysize(.@slot$); .@i++)
			if (getd("RT_R_" + .@slot$[.@i])) .@c++;
		if (!.@c) continue;
		RT_SET[.@n] = .@sid;
		RT_CNT[.@n] = .@c;
		.@n++;
		deletearray .@rw$;
		explode(.@rw$, getvariableofnpc(.s_rw$[.@s], "Rune Tablet#ws"), ",");
		.@given = getd("RT_W_" + .@sid);
		for (.@k = 0; .@k < 7; .@k++) {
			.@item = atoi(.@rw$[.@k]);
			if (!.@item || (.@given & (1 << .@k))) continue;
			if (.@k == 0) .@due = 1;
			else if (.@k == 6) .@due = getarraysize(.@slot$);
			else .@due = .@k + 1;
			if (.@c < .@due) continue;
			getitem .@item, 1;
			.@given |= 1 << .@k;
			dispbottom "Rune Tablet: " + getvariableofnpc(.s_name$[.@s], "Rune Tablet#ws") + " reward: " + getitemname(.@item) + ".";
		}
		setd "RT_W_" + .@sid, .@given;
	}
	if (getarraysize(RT_SET))
		bonus_script "{ callfunc \"F_RuneTabletBonus\"; }", 999999999, 512|1024;
	recalculatestat;
	return;

OnReq:
	.@r$ = .@atcmd_parameters$[0];
	.@cmd$ = .@atcmd_parameters$[1];
	.@arg$ = .@atcmd_parameters$[2];
	.@ok = getvariableofnpc(.many, "Rune Tablet#ws");   // 0 when the Rune Tablet isn't loaded
	if (!.@ok) {
		callsub S_Reply, .@r$, "err|The Rune Tablet isn't on this server.";
		end;
	}
	if (.@cmd$ == "state") {
		.@out$ = "r:";
		.@nr = getarraysize(getvariableofnpc(.r_id, "Rune Tablet#ws"));
		for (.@i = 0; .@i < .@nr; .@i++) {
			.@rid = getvariableofnpc(.r_id[.@i], "Rune Tablet#ws");
			if (getd("RT_R_" + .@rid)) .@out$ += .@rid + ",";
		}
		.@out$ += ";s:";
		.@ns = getarraysize(getvariableofnpc(.s_id, "Rune Tablet#ws"));
		for (.@s = 0; .@s < .@ns; .@s++) {
			.@sid = getvariableofnpc(.s_id[.@s], "Rune Tablet#ws");
			if (getd("RT_L_" + .@sid) || getd("RT_F_" + .@sid))
				.@out$ += .@sid + "=" + getd("RT_L_" + .@sid) + "/" + getd("RT_F_" + .@sid) + ",";
		}
		callsub S_Reply, .@r$, .@out$;
		end;
	}
	if (.@cmd$ == "count") {
		explode(.@id$, .@arg$, ",");
		.@out$ = "";
		for (.@i = 0; .@i < getarraysize(.@id$) && .@i < 60; .@i++)
			.@out$ += atoi(.@id$[.@i]) + "=" + countitem(atoi(.@id$[.@i])) + ",";
		callsub S_Reply, .@r$, .@out$;
		end;
	}
	if (.@cmd$ == "register") {
		.@rid = atoi(.@arg$);
		.@j = inarray(getvariableofnpc(.r_id, "Rune Tablet#ws"), .@rid);
		if (.@j < 0) { callsub S_Reply, .@r$, "err|This rune can't be registered here."; end; }
		if (getd("RT_R_" + .@rid)) { callsub S_Reply, .@r$, "err|Already registered."; end; }
		.@mat$ = getvariableofnpc(.r_mat$[.@j], "Rune Tablet#ws");
		if (!callsub(S_Has, .@mat$)) { callsub S_Reply, .@r$, "err|You don't have all the materials."; end; }
		callsub S_Take, .@mat$;
		setd "RT_R_" + .@rid, 1;
		callsub S_Recount;
		specialeffect2 EF_REFINEOK;
		callsub S_Reply, .@r$, "ok|" + getvariableofnpc(.r_name$[.@j], "Rune Tablet#ws") + " is registered.";
		end;
	}
	if (.@cmd$ == "upgrade") {
		.@sid = atoi(.@arg$);
		.@s = inarray(getvariableofnpc(.s_id, "Rune Tablet#ws"), .@sid);
		if (.@s < 0) { callsub S_Reply, .@r$, "err|This set can't be upgraded here."; end; }
		.@c = 0;
		deletearray .@slot$;
		explode(.@slot$, getvariableofnpc(.s_slot$[.@s], "Rune Tablet#ws"), ",");
		for (.@i = 0; .@i < getarraysize(.@slot$); .@i++)
			if (getd("RT_R_" + .@slot$[.@i])) .@c++;
		if (!.@c) { callsub S_Reply, .@r$, "err|Register at least one of its runes first."; end; }
		.@lv = getd("RT_L_" + .@sid);
		if (.@lv >= 15) { callsub S_Reply, .@r$, "err|It is already +15."; end; }
		deletearray .@up$; deletearray .@rate$; deletearray .@fail$;
		explode(.@up$, getvariableofnpc(.s_up$[.@s], "Rune Tablet#ws"), "|");
		explode(.@rate$, getvariableofnpc(.s_rate$[.@s], "Rune Tablet#ws"), ",");
		explode(.@fail$, getvariableofnpc(.s_fail$[.@s], "Rune Tablet#ws"), ",");
		if (!callsub(S_Has, .@up$[.@lv])) { callsub S_Reply, .@r$, "err|You don't have all the materials."; end; }
		.@chance = atoi(.@rate$[.@lv]) + getd("RT_F_" + .@sid);
		if (.@chance > 100000) .@chance = 100000;
		callsub S_Take, .@up$[.@lv];
		if (rand(100000) < .@chance) {
			setd "RT_L_" + .@sid, .@lv + 1;
			setd "RT_F_" + .@sid, 0;
			specialeffect2 EF_REFINEOK;
			if (getarraysize(RT_SET))
				bonus_script "{ callfunc \"F_RuneTabletBonus\"; }", 999999999, 512|1024;
			recalculatestat;
			callsub S_Reply, .@r$, "ok|success|" + (.@lv + 1);
		} else {
			setd "RT_F_" + .@sid, getd("RT_F_" + .@sid) + atoi(.@fail$[.@lv]);
			specialeffect2 EF_REFINEFAIL;
			callsub S_Reply, .@r$, "ok|fail|" + getd("RT_F_" + .@sid);
		}
		end;
	}
	if (.@cmd$ == "decomplist") {
		getinventorylist;
		.@out$ = "";
		for (.@i = 0; .@i < @inventorylist_count; .@i++) {
			.@id = @inventorylist_id[.@i];
			.@d = inarray(getvariableofnpc(.d_id, "Rune Tablet#ws"), .@id);
			if (.@d < 0 || @inventorylist_equip[.@i] || inarray(.@seen, .@id) >= 0) continue;
			.@seen[getarraysize(.@seen)] = .@id;
			.@out$ += .@id + "=" + countitem(.@id) + ":" + (getvariableofnpc(.d_many$[.@d], "Rune Tablet#ws") != "" ? 1 : 0) + ",";
		}
		callsub S_Reply, .@r$, .@out$;
		end;
	}
	if (.@cmd$ == "decompose") {
		.@id = atoi(.@arg$);
		.@many = getvariableofnpc(.many, "Rune Tablet#ws");
		.@qty = (atoi(.@atcmd_parameters$[3]) > 1) ? .@many : 1;
		.@d = inarray(getvariableofnpc(.d_id, "Rune Tablet#ws"), .@id);
		if (.@d < 0) { callsub S_Reply, .@r$, "err|That can't be decomposed."; end; }
		if (countitem(.@id) < .@qty) { callsub S_Reply, .@r$, "err|You don't carry enough of it."; end; }
		.@rows$ = (.@qty > 1) ? getvariableofnpc(.d_many$[.@d], "Rune Tablet#ws") : getvariableofnpc(.d_one$[.@d], "Rune Tablet#ws");
		if (.@rows$ == "") .@rows$ = getvariableofnpc(.d_one$[.@d], "Rune Tablet#ws");
		delitem .@id, .@qty;
		deletearray .@row$;
		explode(.@row$, .@rows$, "|");
		.@out$ = "ok|";
		for (.@i = 0; .@i < getarraysize(.@row$); .@i++) {
			deletearray .@f$;
			explode(.@f$, .@row$[.@i], ",");
			if (rand(100000) >= atoi(.@f$[3])) continue;
			.@q = rand(atoi(.@f$[1]), atoi(.@f$[2]));
			getitem atoi(.@f$[0]), .@q;
			.@out$ += .@f$[0] + ":" + .@q + ",";
		}
		callsub S_Reply, .@r$, .@out$;
		end;
	}
	callsub S_Reply, .@r$, "err|Unknown request.";
	end;
}
